POST
/lancamentos/ Cria um novo lançamento 
curl -X POST http://127.0.0.1:5000/lancamentos/ \
  -H "Content-Type: application/json" \
  -d '{"descricao":"Pagamento de fornecedor","tipo":"debito","valor":1500.00,"data":"2026-05-15"}'
import requests
response = requests.post("http://127.0.0.1:5000/lancamentos/", json={
    "descricao": "Pagamento de fornecedor",
    "tipo": "debito",
    "valor": 1500.00,
    "data": "2026-05-15"
})
print(response.json())
const res = await fetch("http://127.0.0.1:5000/lancamentos/", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ descricao: "Pagamento de fornecedor", tipo: "debito", valor: 1500.00, data: "2026-05-15" })
});
console.log(await res.json());
POST
/lancamentos/<id>/rateio Define o rateio por centro de custo (soma deve ser 100%) 
curl -X POST http://127.0.0.1:5000/lancamentos/1/rateio \
  -H "Content-Type: application/json" \
  -d '{"rateios":[{"centro_id":1,"percentual":60},{"centro_id":2,"percentual":40}]}'
import requests
response = requests.post("http://127.0.0.1:5000/lancamentos/1/rateio", json={
    "rateios": [
        {"centro_id": 1, "percentual": 60},
        {"centro_id": 2, "percentual": 40}
    ]
})
print(response.json())  # soma deve ser exatamente 100%
const res = await fetch("http://127.0.0.1:5000/lancamentos/1/rateio", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ rateios: [{ centro_id: 1, percentual: 60 }, { centro_id: 2, percentual: 40 }] })
});
console.log(await res.json());
